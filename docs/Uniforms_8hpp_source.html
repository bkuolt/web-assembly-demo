161 [[nodiscard]] bool EnsureCapacity(size_t drawCount) noexcept {
162 if (!device_ || !bgl_ || drawCount == 0) {
163 return false;
164 }
165 if (drawCount <= capacityDraws_ && buffer_ && bindGroup_) {
166 return true;
167 }
168 const size_t newCapacity = std::max<size_t>(64, std::bit_ceil(drawCount));
169 WGPUBufferDescriptor bufDesc = WGPU_BUFFER_DESCRIPTOR_INIT;
170 bufDesc.size = static_cast<uint64_t>(newCapacity) * sizeof(T);
171 bufDesc.usage = WGPUBufferUsage_Uniform | WGPUBufferUsage_CopyDst;
172 bufDesc.label = ToWgpuStringView(label_);
173 BufferHandle newBuf{wgpuDeviceCreateBuffer(device_, &bufDesc)};
174 if (!newBuf) {
175 return false;
176 }
177
178 WGPUBindGroupEntry bgEntry = WGPU_BIND_GROUP_ENTRY_INIT;
179 bgEntry.binding = 0;
180 bgEntry.buffer = newBuf.Get();
181 bgEntry.offset = 0;
182 bgEntry.size = sizeof(T);
183
184 WGPUBindGroupDescriptor bgDesc = WGPU_BIND_GROUP_DESCRIPTOR_INIT;
185 bgDesc.layout = bgl_;
186 bgDesc.entryCount = 1;
187 bgDesc.entries = &bgEntry;
188 bgDesc.label = ToWgpuStringView(label_);
189 BindGroupHandle newBg{wgpuDeviceCreateBindGroup(device_, &bgDesc)};
190 if (!newBg) {
191 return false;
192 }
193
194 bindGroup_ = std::move(newBg);
195 buffer_ = std::move(newBuf);
196 capacityDraws_ = newCapacity;
197 return true;
198 }
199
200 [[nodiscard]] WGPUBindGroup Upload(std::span<const T> items) noexcept {
201 if (items.empty() || !queue_ || !EnsureCapacity(items.size())) {
202 return nullptr;
203 }
204 wgpuQueueWriteBuffer(queue_, buffer_.Get(), 0, items.data(), items.size() * sizeof(T));
205 return bindGroup_.Get();
206 }

212 std::string_view label_{"DynamicUniformRing"};
213 BufferHandle buffer_{};
214 BindGroupHandle bindGroup_{};
215 size_t capacityDraws_{0};
216};
