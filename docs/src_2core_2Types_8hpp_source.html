77 [[nodiscard]] AABB Transformed(const glm::mat4& transform) const noexcept {
78 if (minBound.x > maxBound.x)
79 return AABB{};
80 const glm::vec3 c{Center()};
81 const glm::vec3 e{Extents()};
82 const glm::vec3 newCenter{transform * glm::vec4{c, 1.0f}};
83 const glm::vec3 newExtents{
84 std::abs(transform[0][0]) * e.x + std::abs(transform[1][0]) * e.y + std::abs(transform[2][0]) * e.z,
85 std::abs(transform[0][1]) * e.x + std::abs(transform[1][1]) * e.y + std::abs(transform[2][1]) * e.z,
86 std::abs(transform[0][2]) * e.x + std::abs(transform[1][2]) * e.y + std::abs(transform[2][2]) * e.z};
87 return AABB{.minBound = newCenter - newExtents, .maxBound = newCenter + newExtents};
88 }

97 [[nodiscard]] bool Intersects(const Ray& ray, float& tMinOut, float& tMaxOut) const noexcept {
98 float tmin{0.0f};
99 float tmax{1e30f};
100 for (int i{0}; i < 3; ++i) {
101 const float d =
102 (std::abs(ray.direction[i]) > 1e-8f) ? ray.direction[i] : std::copysign(1e-8f, ray.direction[i]);
103 const float invD{1.0f / d};
104 float t0{(minBound[i] - ray.origin[i]) * invD};
105 float t1{(maxBound[i] - ray.origin[i]) * invD};
106 if (invD < 0.0f)
107 std::swap(t0, t1);
108 tmin = std::max(tmin, t0);
109 tmax = std::min(tmax, t1);
110 if (tmax < tmin)
111 return false;
112 }
113 tMinOut = tmin;
114 tMaxOut = tmax;
115 return true;
116 }

160 static Frustum FromViewProjection(const glm::mat4& vp) noexcept {
161 Frustum f{};
162 // Left plane
163 f.planes[0].normal = glm::vec3{vp[0][3] + vp[0][0], vp[1][3] + vp[1][0], vp[2][3] + vp[2][0]};
164 f.planes[0].distance = vp[3][3] + vp[3][0];
165 // Right plane
166 f.planes[1].normal = glm::vec3{vp[0][3] - vp[0][0], vp[1][3] - vp[1][0], vp[2][3] - vp[2][0]};
167 f.planes[1].distance = vp[3][3] - vp[3][0];
168 // Bottom plane
169 f.planes[2].normal = glm::vec3{vp[0][3] + vp[0][1], vp[1][3] + vp[1][1], vp[2][3] + vp[2][1]};
170 f.planes[2].distance = vp[3][3] + vp[3][1];
171 // Top plane
172 f.planes[3].normal = glm::vec3{vp[0][3] - vp[0][1], vp[1][3] - vp[1][1], vp[2][3] - vp[2][1]};
173 f.planes[3].distance = vp[3][3] - vp[3][1];
174 // Near plane (WebGPU 0..1 depth)
175 f.planes[4].normal = glm::vec3{vp[0][2], vp[1][2], vp[2][2]};
176 f.planes[4].distance = vp[3][2];
177 // Far plane
178 f.planes[5].normal = glm::vec3{vp[0][3] - vp[0][2], vp[1][3] - vp[1][2], vp[2][3] - vp[2][2]};
179 f.planes[5].distance = vp[3][3] - vp[3][2];
180
181 for (auto& plane : f.planes) {
182 plane.Normalize();
183 }
184 return f;
185 }

190 [[nodiscard]] bool IntersectsAABB(const AABB& box) const noexcept {
191 for (const auto& plane : planes) {
192 glm::vec3 p{box.minBound};
193 if (plane.normal.x >= 0.0f)
194 p.x = box.maxBound.x;
195 if (plane.normal.y >= 0.0f)
196 p.y = box.maxBound.y;
197 if (plane.normal.z >= 0.0f)
198 p.z = box.maxBound.z;
199
200 if (plane.DistanceToPoint(p) < 0.0f) {
201 return false;
202 }
203 }
204 return true;
