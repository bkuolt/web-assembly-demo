21[[nodiscard]] inline uint16_t FloatToHalfBits(float val) noexcept {
22 if (std::isnan(val))
23 return 0x7E00u;
24 if (std::isinf(val))
25 return (val < 0.0f) ? 0xFC00u : 0x7C00u;
26 val = std::clamp(val, -65504.0f, 65504.0f);
27 const uint32_t f32{std::bit_cast<uint32_t>(val)};
28 const uint32_t sign{(f32 >> 16u) & 0x8000u};
29 const int32_t exp{static_cast<int32_t>((f32 >> 23u) & 0xFFu) - 127 + 15};
30 uint32_t mant{f32 & 0x007FFFFFu};
31
32 if (exp <= 0) {
33 if (exp < -10)
34 return static_cast<uint16_t>(sign);
35 mant = (mant | 0x00800000u) >> static_cast<uint32_t>(1 - exp);
36 return static_cast<uint16_t>(sign | (mant >> 13u));
37 }
38 if (exp >= 31) {
39 return static_cast<uint16_t>(sign | 0x7C00u);
40 }
41 return static_cast<uint16_t>(sign | (static_cast<uint32_t>(exp) << 10u) | (mant >> 13u));
42}

52[[nodiscard]] inline glm::vec3 CubeFaceDirection(uint32_t faceIndex, float u, float v) noexcept {
53 glm::vec3 dir{0.0f};
54 switch (faceIndex) {
55 case 0:
56 dir = glm::vec3{1.0f, -v, -u};
57 break; // +X
58 case 1:
59 dir = glm::vec3{-1.0f, -v, u};
60 break; // -X
61 case 2:
62 dir = glm::vec3{u, 1.0f, v};
63 break; // +Y
64 case 3:
65 dir = glm::vec3{u, -1.0f, -v};
66 break; // -Y
67 case 4:
68 dir = glm::vec3{u, -v, 1.0f};
69 break; // +Z
70 default:
71 dir = glm::vec3{-u, -v, -1.0f};
72 break; // -Z
73 }
74 return glm::normalize(dir);
75}

77inline void CollectUniquePrimitives(scene::Scene& scene, std::vector<scene::MeshPrimitive*>& outPrims) {
78 outPrims.clear();
79 for (auto& mesh : scene.meshes) {
80 if (mesh) {
81 for (auto& prim : mesh->primitives) {
82 outPrims.push_back(&prim);
83 }
84 }
85 }
86 auto collectNode = [&](auto& self, scene::SceneNode& node) -> void {
87 for (auto& prim : node.GetPrimitives()) {
88 outPrims.push_back(&prim);
89 }
90 for (auto& child : node.children) {
91 if (child) {
92 self(self, *child);
93 }
94 }
95 };
96 for (auto& root : scene.rootNodes) {
97 if (root) {
98 collectNode(collectNode, *root);
99 }
100 }
101 std::sort(outPrims.begin(), outPrims.end());
102 outPrims.erase(std::unique(outPrims.begin(), outPrims.end()), outPrims.end());
103}
