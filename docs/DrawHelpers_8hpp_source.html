33inline void PopulatePerDrawMorphAndSkin(WGPUQueue queue, scene::MeshPrimitive& prim, const scene::SceneNode* node,
34 const glm::mat4& worldMat, const glm::mat4& viewProj,
35 const glm::mat4& lightViewProj, uint32_t baseDrawFlags, float lodTransition,
36 std::span<const scene::Skin> skins,
37 std::span<const std::shared_ptr<scene::SceneNode>> allNodes,
38 uint64_t skinFrameId, WGPUBindGroup dummySkinBg,
39 PreparedDrawBindings& out) noexcept {
40 out.uniforms = PerDrawUniforms{};
41 out.uniforms.viewProj = viewProj;
42 out.uniforms.model = worldMat;
43 out.uniforms.lightViewProj = lightViewProj;
44 out.uniforms.drawParams = glm::vec4{lodTransition, 0.0f, 0.0f, 0.0f};
45 uint32_t flags{baseDrawFlags};
46
47 if (prim.morphDirty && !prim.gpuMorphDeltaBuffer && prim.gpuVertexBuffer && !prim.vertices.empty()) {
48 wgpuQueueWriteBuffer(queue, prim.gpuVertexBuffer, 0, prim.vertices.data(),
49 prim.vertices.size() * sizeof(core::Vertex));
50 prim.morphDirty = false;
51 }
52
53 if (node && prim.gpuMorphDeltaBuffer && prim.gpuMorphTargetCount > 0 && !node->morphWeights.empty()) {
54 const uint32_t count{
55 std::min<uint32_t>({static_cast<uint32_t>(node->morphWeights.size()), prim.gpuMorphTargetCount, 8u})};
56 bool hasActiveMorph{false};
57 for (uint32_t m{0}; m < count; ++m) {
58 const float w{node->morphWeights[m]};
59 if (m < 4u) {
60 out.uniforms.morphWeights0[static_cast<glm::length_t>(m)] = w;
61 } else {
62 out.uniforms.morphWeights1[static_cast<glm::length_t>(m - 4u)] = w;
63 }
64 if (std::abs(w) > 1e-6f) {
65 hasActiveMorph = true;
66 }
67 }
68 if (hasActiveMorph) {
69 out.uniforms.morphTargetCount = count;
70 out.uniforms.vertexCount = prim.gpuMorphVertexCount;
71 flags |= MaterialFlags::HasMorphTargets;
72 }
73 }
74
75 out.skinBindGroup = dummySkinBg;
76 if (node && node->skinIndex >= 0 && static_cast<size_t>(node->skinIndex) < skins.size()) {
77 const auto& skin = skins[static_cast<size_t>(node->skinIndex)];
78 if (!skin.joints.empty() && skin.gpuJointBuffer && skin.gpuBindGroup) {
79 const auto jointsSpan = skin.GetOrComputeJointMatrices(skinFrameId, node, worldMat, allNodes);
80 if (!jointsSpan.empty()) {
81 if (skin.NeedsGpuUpload(skinFrameId, node)) {
82 wgpuQueueWriteBuffer(queue, skin.gpuJointBuffer, 0, jointsSpan.data(),
83 jointsSpan.size() * sizeof(glm::mat4));
84 skin.MarkGpuUploaded(skinFrameId, node);
85 }
86 out.uniforms.jointCount = static_cast<uint32_t>(jointsSpan.size());
87 out.skinBindGroup = skin.gpuBindGroup;
88 flags |= MaterialFlags::HasSkinning;
89 }
90 }
91 }
92
93 out.uniforms.drawFlags = flags;
94}

175 [[nodiscard]] core::Result<void> Initialize(WGPUDevice device, size_t initialInstanceCapacity = 256,
176 std::string_view label = "DynamicInstanceBuffer") {
177 if (!device) {
178 return std::unexpected("Cannot initialize DynamicInstanceBuffer with null device");
179 }
180 device_ = device;
181 label_ = std::string(label);
182 const size_t minCapacity = std::max<size_t>(initialInstanceCapacity, 1);
183 return EnsureCapacityBytes(nullptr, minCapacity * sizeof(glm::mat4), true);
184 }

186 void Shutdown() noexcept {
187 ReleaseBuffer();
188 device_ = nullptr;
189 capacityBytes_ = 0;
190 identityWritten_ = false;
191 }

193 void Upload(WGPUQueue queue, std::span<const glm::mat4> instances) {
194 if (!device_ || !queue) {
195 return;
196 }
197 const size_t requiredBytes = (1u + instances.size()) * sizeof(glm::mat4);
198 const bool resized = !buffer_ || requiredBytes > capacityBytes_;
199 if (resized) {
200 size_t nextBytes = std::max<size_t>(sizeof(glm::mat4) * 64u, capacityBytes_ == 0 ? 16384u : capacityBytes_);
201 while (nextBytes < requiredBytes) {
202 nextBytes *= 2u;
203 }
204 if (!EnsureCapacityBytes(queue, nextBytes, true)) {
205 return;
206 }
207 } else if (!identityWritten_) {
208 const glm::mat4 identity{1.0f};
209 wgpuQueueWriteBuffer(queue, buffer_.Get(), 0, &identity, sizeof(glm::mat4));
210 identityWritten_ = true;
211 }
212
213 if (!instances.empty() && buffer_) {
214 wgpuQueueWriteBuffer(queue, buffer_.Get(), sizeof(glm::mat4), instances.data(),
215 instances.size() * sizeof(glm::mat4));
216 }
217 }

220 [[nodiscard]] size_t GetCapacityBytes() const noexcept { return capacityBytes_; }

253 [[nodiscard]] core::Result<void> EnsureCapacityBytes(WGPUQueue queue, size_t newCapacityBytes,
254 bool mappedIdentity) {
255 ReleaseBuffer();
256 WGPUBufferDescriptor desc = WGPU_BUFFER_DESCRIPTOR_INIT;
257 desc.usage = WGPUBufferUsage_Vertex | WGPUBufferUsage_CopyDst;
258 desc.size = newCapacityBytes;
259 desc.label = ToWgpuStringView(label_);
260 desc.mappedAtCreation = (mappedIdentity && queue == nullptr) ? 1u : 0u;
261
262 BufferHandle nextBuf{wgpuDeviceCreateBuffer(device_, &desc)};
263 if (!nextBuf) {
264 return std::unexpected("Failed to create DynamicInstanceBuffer");
265 }
266 if (desc.mappedAtCreation) {
267 void* mapped = wgpuBufferGetMappedRange(nextBuf.Get(), 0, sizeof(glm::mat4));
268 if (mapped) {
269 const glm::mat4 identity{1.0f};
270 std::memcpy(mapped, &identity, sizeof(glm::mat4));
271 }
272 wgpuBufferUnmap(nextBuf.Get());
273 identityWritten_ = true;
274 } else if (queue != nullptr) {
275 const glm::mat4 identity{1.0f};
276 wgpuQueueWriteBuffer(queue, nextBuf.Get(), 0, &identity, sizeof(glm::mat4));
277 identityWritten_ = true;
278 }
279
280 core::MemoryTracker::Instance().TrackBufferAlloc(newCapacityBytes);
281 buffer_ = std::move(nextBuf);
282 capacityBytes_ = newCapacityBytes;
283 return {};
284 }
